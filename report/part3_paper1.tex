\subsection{Phân tích bài báo: Active Bipartite Ranking}
\textbf{Thông tin nghiên cứu:} Bài báo của James Cheshire, Vincent Laurent, và Stephan Clémençon công bố tại Hội nghị NeurIPS (2023).

\subsubsection{Vấn đề}


\begin{tcolorbox}\textbf{Câu hỏi nghiên cứu trung tâm:}\bfseries
  \textit{Làm thế nào để xây dựng một thuật toán Học chủ động (Active Learning) chuyên biệt cho bài toán xếp hạng hai phía (Bipartite Ranking), cho phép mô hình tự động lựa chọn và chỉ yêu cầu gán nhãn những điểm dữ liệu "khó phân định" nhất, nhằm xấp xỉ đường cong ROC tối ưu với chi phí gán nhãn tối thiểu?}
\end{tcolorbox}
\medskip

Động lực của bài báo xuất phát từ những điểm nghẽn thực tế:
\begin{enumerate}
  \item Trong các hệ thống phát hiện gian lận, chấm điểm tín dụng hay chẩn đoán y khoa, việc thu thập dữ liệu rác (unlabeled) thì rẻ, nhưng chi phí để một chuyên gia xác nhận nhãn (labeling) là vô cùng đắt đỏ.
  \item Active Learning đã rất phát triển cho bài toán phân loại nhị phân, nhưng lại gần như vắng bóng trong bài toán Xếp hạng. Lý do là vì phân loại chỉ quan tâm đến ranh giới cục bộ (local boundary), trong khi xếp hạng mang tính \textbf{toàn cục (global nature)} -- vị trí của một điểm bị ảnh hưởng bởi mối quan hệ tương đối của nó với tất cả các điểm còn lại.
\end{enumerate}

\subsubsection{Đóng góp chính}
Bài báo đóng góp 4 kết quả nền tảng cho lý thuyết Học chủ động trong Xếp hạng:
\begin{enumerate}
  \item \textbf{Cơ sở toán học chuẩn xác:} Định hình khuôn khổ học chủ động cho Bipartite Ranking thông qua việc tối ưu hóa khoảng cách chuẩn vô cùng (sup norm) giữa đường cong ROC thực nghiệm và ROC lý tưởng.
  \item \textbf{Thuật toán Active-Rank:} Đề xuất một chiến lược lấy mẫu chọn lọc (selective sampling) thiết kế riêng cho bài toán xếp hạng.
  \item \textbf{Đảm bảo PAC (Probably Approximately Correct):} Chứng minh thuật toán đạt chuẩn hội tụ $\text{PAC}(\epsilon, \delta)$ với ranh giới sai số chặt chẽ.
  \item \textbf{Phân tích độ phức tạp lấy mẫu (Sampling Complexity):} Thiết lập cả ranh giới trên (upper bound) và ranh giới dưới (lower bound) cho thời gian lấy mẫu kỳ vọng của bất kỳ thuật toán PAC nào trong bài toán này.
\end{enumerate}

\subsubsection{Thiết lập và ký hiệu}
% ============================================================
\paragraph{Mô hình xác suất và Đường cong ROC tối ưu}
Cho không gian dữ liệu $\mathcal{X}$ và nhãn $\mathcal{Y} = \{0, 1\}$. Đặt $p = \mathbb{P}(Y=1)$ là xác suất của lớp tích cực. Hàm xác suất hậu nghiệm (conditional probability) được định nghĩa là:
\begin{equation}
  \eta(x) = \mathbb{P}(Y=1 \mid X=x)
\end{equation}

Đối với một hàm cho điểm (scoring function) $H: \mathcal{X} \to \mathbb{R}$, hiệu năng xếp hạng của nó được đánh giá trọn vẹn qua đường cong ROC (Receiver Operating Characteristic), ký hiệu là $ROC_H(\cdot)$. 

\begin{dinh_nghia}[Đường cong ROC tối ưu (Optimal ROC Curve)]
  \label{optimal_roc}
  Đường cong ROC đạt giá trị cực đại tại mọi điểm (được gọi là $ROC^*$) đạt được khi và chỉ khi hàm cho điểm $H$ là một phép biến đổi tăng ngặt (strictly increasing transformation) của hàm xác suất hậu nghiệm $\eta(x)$. Tức là $H^*(x) = \eta(x)$.
\end{dinh_nghia}

Mục tiêu của khung học chủ động là xây dựng một hàm $\hat{H}$ từ một tập các truy vấn gán nhãn nhỏ nhất sao cho khoảng cách giữa đường cong ROC của nó và ROC tối ưu bị chặn bởi $\epsilon$:
\begin{equation}
  \| ROC_{\hat{H}} - ROC^* \|_{\infty} = \sup_{t \in [0,1]} | ROC_{\hat{H}}(t) - ROC^*(t) | \le \epsilon
  \label{eq:roc_sup_norm}
\end{equation}

\begin{lienhe_ly_thuyet}
  \textbf{Sự kế thừa và mở rộng từ Chương 9 [FML]:} \\
  Trong \textbf{Mục 9.5.2} của giáo trình Foundations of Machine Learning, tác giả định nghĩa và đánh giá Bipartite Ranking thông qua chỉ số \textbf{AUC} (Area Under the ROC Curve) -- vốn là một giá trị vô hướng (scalar summary) tích phân từ đường cong ROC.
  
  Bài báo thay vì chỉ tối ưu diện tích AUC (phần không gian dưới đường cong), bài báo hướng tới việc xấp xỉ \textbf{toàn bộ hình dáng của đường cong ROC} thông qua chuẩn vô cùng $\| \cdot \|_{\infty}$ (phương trình \ref{eq:roc_sup_norm}). Điều này đảm bảo rằng mô hình không chỉ tốt về mặt tổng thể (AUC) mà còn chính xác ở mọi ngưỡng quyết định (thresholds) trong thực tế.
\end{lienhe_ly_thuyet}

% ============================================================
\subsubsection{Chiến lược Lấy mẫu Chủ động (Active-Rank)}
% ============================================================
\paragraph{Nghịch lý của Active Learning trong Xếp hạng}
Trong phân loại nhị phân, thuật toán Active Learning chỉ cần tập trung yêu cầu dán nhãn các điểm nằm gần ranh giới quyết định (với $\eta(x) \approx 0.5$). Tuy nhiên, đối với xếp hạng, một điểm có $\eta(x) = 0.9$ và một điểm có $\eta(x) = 0.8$ đều được phân loại là $1$, nhưng hệ thống bắt buộc phải nhận diện được sự khác biệt giữa chúng để xếp điểm $0.9$ lên trên điểm $0.8$. Do đó, mô hình phải học xấp xỉ hàm $\eta(x)$ trên \textbf{toàn bộ không gian dữ liệu} chứ không chỉ ở vùng biên.

\begin{bo_de}[Sự phân hoạch không gian (Space Partitioning)]
  \label{active_partition}
  Thuật toán Active-Rank chia không gian $\mathcal{X}$ thành các vùng (regions) $C_m$. Tại mỗi vùng, thuật toán duy trì một khoảng tin cậy (confidence interval) $[L_m, U_m]$ cho giá trị thực của $\eta(x)$. Nếu độ rộng của khoảng này, $U_m - L_m$, đủ lớn để gây ra sai số vượt quá $\epsilon$ đối với đường cong ROC, vùng $C_m$ sẽ được đánh dấu là "vùng tích cực" (active region) và thuật toán sẽ yêu cầu chuyên gia gán nhãn các mẫu rơi vào vùng này.
\end{bo_de}

\begin{dinh_ly}[Đảm bảo hội tụ PAC cho Active-Rank]
  \label{thm_pac_active_rank}
  Với bất kỳ tham số dung sai $\epsilon > 0$ và độ tin cậy $\delta \in (0,1)$, thuật toán \texttt{active-rank} dừng lại sau một số hữu hạn các truy vấn và trả về một hàm xếp hạng $\hat{H}$ thỏa mãn:
  \begin{equation}
    \mathbb{P} \left( \| ROC_{\hat{H}} - ROC^* \|_{\infty} \le \epsilon \right) \ge 1 - \delta
  \end{equation}
\end{dinh_ly}

\begin{nhan_xet_box}
  \textbf{Ý nghĩa độ phức tạp thời gian lấy mẫu (Sampling Complexity):} \\
  Bài báo đã chứng minh được rằng, thời gian kỳ vọng để thuật toán dừng (số lượng nhãn cần hỏi) tỷ lệ thuận với $\mathcal{O}\left( \log(1/\delta) \right)$ và phụ thuộc mạnh vào cấu trúc phân phối của $\eta(X)$. Nếu dữ liệu có nhiều khoảng $\eta(x)$ tách biệt rõ ràng, thuật toán sẽ hội tụ cực kỳ nhanh, tiết kiệm đến hơn $80\%$ chi phí dán nhãn so với việc học thụ động (passive learning) sử dụng toàn bộ tập dữ liệu (đã phân tích ở bài báo ICML 2011).
\end{nhan_xet_box}

\subsubsection{Thực nghiệm kiểm chứng}
% ============================================================
Nhóm tác giả tiến hành thực nghiệm thuật toán \texttt{active-rank} trên các tập dữ liệu tổng hợp với cấu trúc phân phối $\eta(x)$ được thiết kế từ đơn giản đến phức tạp (như Mixture of Gaussians).
Mô hình được so sánh với 2 chiến lược cơ sở:
\begin{enumerate}
  \item \textbf{Passive (Thụ động):} Lấy mẫu ngẫu nhiên không có chiến lược (tương tự các thuật toán ở Chương 9).
  \item \textbf{Active-Class (Chủ động cho Phân loại):} Sử dụng thuật toán học chủ động truyền thống vốn chỉ nhắm vào việc tối ưu $0/1$ loss.
\end{enumerate}

\textbf{Kết quả nổi bật:}
Thuật toán \texttt{active-rank} vượt trội hoàn toàn. Để đạt được cùng một mức khoảng cách chuẩn vô cùng $\| ROC_{\hat{H}} - ROC^* \|_{\infty} \approx 0.05$, thuật toán Passive cần tới $\sim 10^5$ mẫu gán nhãn, trong khi \texttt{active-rank} chỉ cần khoảng $\sim 1.5 \times 10^4$ mẫu. Hơn thế nữa, Active-Class thất bại hoàn toàn trong việc thu hẹp khoảng cách ROC vì nó bỏ qua cấu trúc thứ tự toàn cục của dữ liệu.

\subsubsection{Đánh giá phê bình và câu hỏi mở}
% ============================================================
\paragraph{Điểm mạnh}
\begin{enumerate}
  \item Khỏa lấp được một lỗ hổng rất lớn trong lý thuyết học máy: Đưa khung học chủ động (Active Learning) vào bài toán Xếp hạng (Ranking) một cách chuẩn xác về mặt toán học.
  \item Ranh giới lý thuyết rất chặt chẽ, thiết lập được cả giới hạn trên và dưới cho độ phức tạp lấy mẫu.
\end{enumerate}

\paragraph{Hạn chế}
\begin{enumerate}
  \item Thuật toán đề xuất yêu cầu sự phân hoạch (partitioning) không gian dữ liệu. Trong các bài toán thực tế với số chiều đặc trưng (features) rất lớn, việc chia lưới không gian (grid partitioning) sẽ gặp hiện tượng "lời nguyền số chiều" (curse of dimensionality), làm giảm tính khả thi.
  \item Thuật toán chủ yếu nhắm vào môi trường Bipartite (2 lớp).
\end{enumerate}

\begin{tcolorbox}\textbf{Hướng phát triển và Câu hỏi mở:}\bfseries
  \begin{enumerate}
    \item \textit{Mở rộng cho Preference-based Ranking?} Lý thuyết hiện tại mới giải quyết cho Bipartite (Điểm số), liệu có thể áp dụng Active Learning cho cài đặt học dựa trên Sở thích (Mục 9.6 của giáo trình) không?
    \item \textit{Deep Active Ranking?} Có thể kết hợp cơ chế lấy mẫu này với Mạng nơ-ron sâu (Deep Neural Networks) để tự động hóa việc chia vùng thay vì phân hoạch thủ công không?
  \end{enumerate}
\end{tcolorbox}

\subsubsection{Sinh viên Demo thực nghiệm}
% ============================================================
\paragraph{Mục tiêu demo:}
Nhằm kiểm chứng trực quan luận điểm cốt lõi của bài báo, ta xây dựng một chương trình mô phỏng so sánh hiệu năng của chiến lược \texttt{Active-Rank} với chiến lược học thụ động (Passive / Random Sampling) trong cùng một điều kiện ngân sách gán nhãn giới hạn.

\paragraph{Thiết lập thực nghiệm:}
\begin{itemize}
  \item \textbf{Dữ liệu:} Sinh tổng hợp 1000 mẫu hai chiều từ hai phân phối Gauss khác nhau (lớp dương và lớp âm), mô phỏng một ``Unlabeled Pool'' lớn chưa được gán nhãn.
  \item \textbf{Ngân sách (Budget):} Mỗi mô hình chỉ được phép yêu cầu gán nhãn tối đa \textbf{60 mẫu} (tương đương 6\% tổng dữ liệu).
  \item \textbf{Chiến lược Thụ động (Passive):} Chọn ngẫu nhiên 60 mẫu để gán nhãn.
  \item \textbf{Chiến lược Chủ động (Active-Rank):} Xuất phát từ 10 mẫu mồi ngẫu nhiên, sau đó lặp lại việc huấn luyện mô hình hiện tại, dự đoán xác suất hậu nghiệm $\eta(x)$ trên toàn bộ pool, và ưu tiên chọn các mẫu có độ bất định cao nhất (gần ngưỡng 0.5) để yêu cầu chuyên gia gán nhãn. Quá trình này lặp lại cho đến khi sử dụng hết ngân sách.
\end{itemize}

\paragraph{Kết quả và hình ảnh minh họa:}
Hình \ref{fig:roc_demo} so sánh đường cong ROC của hai chiến lược sau khi đã sử dụng hết 60 nhãn. Kết quả cho thấy:
\begin{itemize}
  \item Chiến lược \texttt{Active-Rank} đạt AUC cao hơn rõ rệt (0.8457 so với 0.8418 của Passive).
  \item Đường cong ROC của Active-Rank bám sát góc trên bên trái hơn, chứng tỏ khả năng phân biệt thứ hạng tốt hơn với cùng một lượng nhãn ít ỏi.
  \item Điều này minh chứng cho luận điểm của bài báo: việc chủ động lấy mẫu ở các vùng ``mập mờ'' (gần $\eta(x)=0.5$) giúp cải thiện toàn bộ hình dáng đường cong ROC nhanh hơn so với lấy mẫu ngẫu nhiên.
\end{itemize}

\paragraph{Nhận xét và kết luận từ demo:}
Thực nghiệm đã tái hiện thành công thông điệp chính của nghiên cứu:
\begin{enumerate}
  \item \textbf{Tính khả thi:} Học chủ động hoàn toàn có thể áp dụng hiệu quả vào bài toán xếp hạng hai phía (bipartite ranking) mà không cần đến toàn bộ dữ liệu đã gán nhãn.
  \item \textbf{Ưu thế về chi phí:} Với cùng một ngân sách nhãn hạn chế, Active-Rank cho chất lượng xếp hạng (AUC và hình dáng ROC) vượt trội so với lấy mẫu ngẫu nhiên.
  \item \textbf{Hỗ trợ lý thuyết PAC:} Kết quả thực nghiệm phù hợp với đảm bảo hội tụ lý thuyết mà bài báo đã chứng minh: chỉ cần một lượng mẫu chủ động có kích thước $\mathcal{O}(\log(1/\delta))$ là đủ để xấp xỉ ROC tối ưu với sai số $\epsilon$.
\end{enumerate}

Demo này cũng chỉ ra một hạn chế thực tế: khi số chiều dữ liệu lớn, việc phân hoạch không gian để xác định ``vùng tích cực'' trở nên khó khăn, mở ra hướng phát triển kết hợp với các phương pháp học sâu (deep active ranking) trong tương lai.
